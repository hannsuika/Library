例题：请你编写程序，将二进制转换为八进制并输出~

代码：

#include <iostream>
using namespace std;
int main() 
{
    string s;
    cin >> s;
    while (s.length() % 3)
    {
        s='0'+s;
    }
    for (int i = 2; i < s.length(); i += 3)
    {
        int x = 0; 
        x += (s[i-2]-'0')*4;
        x += (s[i-1]-'0')*2;
        x += (s[i]-'0');
        cout << x;
    }
    return 0;
}
