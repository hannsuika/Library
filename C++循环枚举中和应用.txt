例题：给定两个数 a 和 b，找到一个数字 c，满足：
b 是 a 和 c 的最大公因数；
c≠b；
找到的 c 要尽可能小。（题目满足c < 1000）

题目保证测试数据中a > b;

代码：

#include <iostream>
using namespace std;
int gcd(int a, int b)
{
    int ans = 0;
    for(int i = 1; i <= b; i++)
    {
        if(a % i == 0 && b % i == 0)
        {
            ans = i;
        }
    }
    return ans;
}
int main()
{
    /*
    遍历从 b+1 到 1000，寻找正整数 c；
    判断 gcd(a,c)是否等于b
    如果满足条件，输出结果，结束循环。
    */
    int a, b;
    cin >> a >> b;
    for(int i = b+1; i <1000; i++ ){
        if( gcd(i,a)==b       ){
            cout<<i;
            break;
        }
    }
    return 0;
}
