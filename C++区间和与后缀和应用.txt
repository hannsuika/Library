例题：现有 n 个整数，a[1], a[2] ... a[n]。需要在数组中找到一个平衡点下标，使得平衡点左侧的数之和等于平衡点右侧的数之和。

(1 ≤ n,m <=100000, 1 ≤ k ≤ n, 1 ≤ a[i] ≤ 1000)

代码：

#include <iostream>
using namespace std;
int n;
int a[100010];
int sum1[100010]; //sum1为前缀和数组
int sum2[100010]; //sum2为后缀和数组
int main()
{
    cin>>n;
    for (int i=1;i<=n;i++)
    {
        cin>>a[i];
        sum1[i]=sum1[i-1]+a[i];
    }
    for (int i=n;i>=1;i--)
    {
        sum2[i]=sum2[i+1]+a[i];
    }
    for (int k=2;k<=n-1;k++)
    {
        if (sum1[k]==sum2[k])
        {
            cout<<k;
        }
    }
    
    return 0;
}
